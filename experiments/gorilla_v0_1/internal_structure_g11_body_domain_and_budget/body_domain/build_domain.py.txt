def xhits(tris,y,z):
 a=tris[:,0,1:];b=tris[:,1,1:]-a;c=tris[:,2,1:]-a
 d=np.array([y,z])-a; det=b[:,0]*c[:,1]-b[:,1]*c[:,0]
 ok=np.abs(det)>1e-14
 u=np.divide(d[:,0]*c[:,1]-d[:,1]*c[:,0],det,out=np.zeros(len(det)),where=ok)
 w=np.divide(b[:,0]*d[:,1]-b[:,1]*d[:,0],det,out=np.zeros(len(det)),where=ok)
 ok&=(u>=-1e-9)&(w>=-1e-9)&(u+w<=1+1e-9)
 x=tris[:,0,0]+u*(tris[:,1,0]-tris[:,0,0])+w*(tris[:,2,0]-tris[:,0,0])
 return np.unique(np.round(x[ok],12))
def profile(z,tri):
 # Full loft is independent. Actual front/rear rays constrain x; side closure is NOT certified cavity.
 ys=np.linspace(-.34,.34,17); rows=[];front=[];rear=[]
 for y in ys:
  h=xhits(tri,y,z); f=h[h>.02];r=h[h<-.06]
  front.append(float(max(f)) if len(f) else np.nan)
  rear.append(float(min(r)) if len(r) else np.nan)
  rows.append({'z_m':float(z),'y_m':float(y),'actual_outer_x_hits_m':h.tolist(),'front_actual_hit':bool(len(f)),'rear_actual_hit':bool(len(r))})
 def fill(a,label):
  a=np.array(a); good=np.flatnonzero(np.isfinite(a))
  if len(good)<2:raise RuntimeError('insufficient actual '+label+' surface')
  return np.interp(np.arange(len(a)),good,a[good])
 ff=fill(front,'front');rr=fill(rear,'rear')
 for i,r in enumerate(rows):
  r.update({'front_candidate_x_m':float(ff[i]-.025),'rear_candidate_x_m':float(rr[i]+.025),'front_minus_actual_outer_m':None if np.isnan(front[i]) else -.025,'rear_minus_actual_outer_m':None if np.isnan(rear[i]) else .025,'interpolated_open_source_unknown':bool(np.isnan(front[i]) or np.isnan(rear[i]))})
 ring=[[float(ff[i]-.025),float(y),float(z)] for i,y in enumerate(ys)] + [[float(rr[i]+.025),float(y),float(z)] for i,y in reversed(list(enumerate(ys)))]
 return ring,rows
def loft_native(rings):
 v=np.array(rings); n=v.shape[1]; v=v.reshape(-1,3);f=[]
 # shapely ear clipping handles concave caps without a false fan.
 from shapely.geometry import Polygon
 from shapely.ops import triangulate
 for ringindex in [0,len(rings)-1]:
  poly=Polygon(np.asarray(rings[ringindex])[:,:2]); rr=np.asarray(rings[ringindex])[:,:2]
  for t in triangulate(poly):
   if not poly.covers(t):continue
   q=np.array(t.exterior.coords)[:3];ids=[int(np.argmin(np.linalg.norm(rr-a,axis=1)))+ringindex*n for a in q]
   # Explicit cap orientation; do not use a normals fixer that could alter cavities.
   q3=v[ids]; cz=np.cross(q3[1]-q3[0],q3[2]-q3[0])[2]
   if (ringindex==0 and cz>0) or (ringindex>0 and cz<0):ids=ids[::-1]
   f.append(ids)
 for k in range(len(rings)-1):
  for i in range(n):
   a=k*n+i;b=k*n+(i+1)%n;c=(k+1)*n+(i+1)%n;d=(k+1)*n+i
   f.extend([[a,b,c],[a,c,d]])
 m=mf.Manifold(mf.Mesh64(np.asarray(v),np.asarray(f,np.uint64)))
 if str(m.status())!='Error.NoError':raise RuntimeError(str(m.status()))
 return m

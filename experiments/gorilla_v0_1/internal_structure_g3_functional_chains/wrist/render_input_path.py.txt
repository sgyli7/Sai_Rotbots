from pathlib import Path
import json,runpy
import numpy as np

B=Path(__file__).resolve().parent;n=runpy.run_path(str(B/'screen_wrist_layout.py'),run_name='native_input_view');s=json.loads((B/'selected_installation_scene.json').read_text())
im=Image.new('RGB',(1800,1100),'white');d=ImageDraw.Draw(im);d.text((20,15),'G3 selected native input paths / OEM stock wire + finite custom parts. Conditional interfaces, no assembly acceptance.',fill='black');scale=2500

 key=j['id'].removeprefix('left_');a=np.array(j['axis_world'])*j['input_face_B_sign'];u=np.cross(a,[0,0,1.])

 # nothing is claimed hidden or collisionexcluded by this diagnostic rendering choice.

  if p['role']=='component_module_envelope':continue
  vs=np.array(p['vertices_world_m']);col=(81,114,126) if '_input_' in p['name'] or '_rotor_' in p['name'] or 'WG_' in p['name'] else (139,154,161)
  for face in p['faces']:d.polygon([P(vs[i]) for i in face],fill=col,outline=(35,48,53))
 for p in ps:
  if p['role']!='component_module_envelope':continue
  vs=np.array(p['vertices_world_m']);col=(168,105,47)
  for face in p['faces']:q=[P(vs[i]) for i in face];d.line(q+[q[0]],fill=col,width=1)
 d.text((30+k*580,100),key+'; same2500px/m',fill='black')
 d.text((30+k*580,930),'14mm shaft /5mm key /6mm OWN depth',fill='black')
 d.text((30+k*580,955),'caseC != WG-end B; bare source mass once',fill='black')
im.save(B/'native_input_path_selected.png')

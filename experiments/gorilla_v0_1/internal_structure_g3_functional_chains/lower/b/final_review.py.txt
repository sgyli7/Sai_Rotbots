def M(p):
    return mf.Manifold(mf.Mesh64(np.asarray(p['vertices_world_m'], dtype=np.float64, order='C'),
                                 np.asarray(p['faces'], dtype=np.uint64, order='C')))

def mesh_of(m):
    q = m.to_mesh64()
    return tr.Trimesh(vertices=q.vert_properties[:, :3], faces=q.tri_verts, process=False)

def moments(m):
    t = mesh_of(m)
    p = t.mass_properties
    assert abs(p.volume - m.volume()) < 1e-8
    return {'volume_m3': float(p.volume), 'com_world_m': p.center_mass.tolist(),
            'inertia_at_com_per_density_kgm2': p.inertia.tolist()}

def scaled(unit, masses):
    v = unit['volume_m3']
    return {**unit, 'mass_range_kg': [float(x) for x in masses],
            'inertia_at_com_kg_m2_range': [(np.array(unit['inertia_at_com_per_density_kgm2']) * m / v).tolist() for m in masses]}

def aggregate(rows, label):
    out = []
    for k in range(3):
        mass = sum(x['mass_range_kg'][k] for x in rows)
        com = sum(np.array(x['com_world_m']) * x['mass_range_kg'][k] for x in rows) / mass
        inertia = np.zeros((3, 3))
        for x in rows:
            d = np.array(x['com_world_m']) - com
            inertia += np.array(x['inertia_at_com_kg_m2_range'][k]) + x['mass_range_kg'][k] * (np.dot(d, d) * np.eye(3) - np.outer(d, d))
        out.append({'case': ['low', 'nominal', 'high'][k], 'mass_kg': mass,
                    'com_world_m': com.tolist(), 'inertia_at_com_world_kg_m2': inertia.tolist(),
                    'inertia_eigenvalues_kg_m2': np.linalg.eigvalsh(inertia).tolist()})
    return {'label': label, 'conditions': 'Conditional finite geometry and explicit mass assumptions; collisions/source rejection prevent SI release.', 'range': out}

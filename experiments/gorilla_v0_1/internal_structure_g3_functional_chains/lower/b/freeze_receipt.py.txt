for path, digest in (expected_history | scene['input_hashes']).items():
    got = sha(ROOT / path)
    checks.append({'path': path, 'expected_sha256': digest, 'observed_sha256': got, 'exact_match': got == digest})
assert all(r['exact_match'] for r in checks)
assert report['source_sha256'] == images['source_sha256'] == expected_history['.scratch/gorilla_internal_g3_foot_chain/b/chain_scene.json']
assert report['review_entry_sha256'] == sha(O / 'final_review.py')

for q in report['complete_submodule_with_new_armor_before_unknowns']['range']:

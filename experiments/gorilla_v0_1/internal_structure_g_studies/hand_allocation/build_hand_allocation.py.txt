def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def triangles(f):return np.asarray([[ff[0],ff[i],ff[i+1]] for ff in f for i in range(1,len(ff)-1)],dtype=np.uint32)
def native(p):return trimesh.Trimesh(np.asarray(p['vertices_world_m']),triangles(p['faces']),process=False)
def center(p):
 t=np.asarray(p['vertices_world_m'])[triangles(p['faces'])]
 vv=np.einsum('ij,ij->i',t[:,0],np.cross(t[:,1],t[:,2]))/6
 return ((t.sum(1)/4)*vv[:,None]).sum(0)/vv.sum()
def pinaxis(p):
 v=np.asarray(p['vertices_world_m']);f=max(p['faces'],key=len)
 a=np.cross(v[f[1]]-v[f[0]],v[f[2]]-v[f[0]])
 return a/np.linalg.norm(a)
def mani(mesh):return mf.Manifold(mf.Mesh64(np.asarray(mesh.vertices,dtype=np.float64),np.asarray(mesh.faces,dtype=np.uint32)))
def from_m(m):
 a=m.to_mesh64();return trimesh.Trimesh(a.vert_properties[:,:3],a.tri_verts,process=False)
def box(c,d):
 m=trimesh.creation.box(extents=d);m.apply_translation(c);return m
def cyl(c,r,L,a=(0,0,1)):
 m=trimesh.creation.cylinder(radius=r,height=L,sections=24)
 m.apply_transform(trimesh.geometry.align_vectors([0,0,1],a));m.apply_translation(c);return m
def tube(c,ro,ri,L,a=(0,0,1)):
 return from_m(mani(cyl(c,ro,L,a))-mani(cyl(c,ri,L+.004,a)))
def segment(p,q,ro,ri=0):

def collision_screen(scene):
 # Scope is selected arm/hand reference geometry, not whole robot or dynamics.
 reference=[p for p in base['parts'] if any(p['body'].startswith(s) for s in ['left_forearm','right_forearm','left_palm','right_palm','left_finger','right_finger','left_thumb','right_thumb'])]
 known=[p for p in old['parts'] if p['name'].startswith(('ds_left_forearm_roll_','ds_right_forearm_roll_','ds_left_wrist_','ds_right_wrist_'))]
 targets=reference+known; cache={};hits=[];unknown=[]
 for p in scene['parts']:
  pm=native(p);pb=pm.bounds;P=None
  for q in targets:
   if p['name'].split('_')[1]!=q['name'].split('_')[0] and ('left' in p['name'])!=('left' in q['name']):continue
   qm=native(q);qb=qm.bounds
   if np.any(np.minimum(pb[1],qb[1])-np.maximum(pb[0],qb[0])<=1e-8):continue
   try:
    if P is None:P=mani(pm)
    if q['name'] not in cache:cache[q['name']]=mani(qm)
    Q=cache[q['name']]
    if P.status()!=mf.Error.NoError or Q.status()!=mf.Error.NoError:
     unknown.append([p['name'],q['name'],'source native Manifold invalid, not zero overlap']);continue
    vol=(P^Q).volume()
    if vol>1e-9:hits.append(dict(new=p['name'],reference=q['name'],volume_m3=vol,reference_scope='C15 original' if q in reference else 'E2a old motor/gear module'))
   except Exception as e:unknown.append([p['name'],q['name'],str(e)])
 return dict(scope='neutral new complete hand parts vs actual C15 forearm/palm/finger/thumb and E2a old wrist/forearm module envelopes only; excludes new-new pairs and full robot and all poses',reference_parts=len(targets),new_parts=len(scene['parts']),positive_hits=len(hits),hits=sorted(hits,key=lambda x:x['volume_m3'],reverse=True),unknown_pairs=unknown,accepted=False)

def route_motion(routes):
 joints={j['id']:j for j in old['joint_supports']};out=[]
 for r in routes:
  pts=np.array(r['points_world_m']);nom=r['length_m'];rows=[]
  # Parent driver to downstream anchors: finite true eye paths, still need flexible sheath routing.
  for joint in r['crossed_joints']:
   j=joints[joint];c=np.array(j['center_world_m']);a=np.array(j['axis_world']);a/=np.linalg.norm(a)
   for angle in [-math.pi/4,math.pi/4]:
    moved=pts.copy();R=trimesh.transformations.rotation_matrix(angle,a,c)
    moved[2:]=trimesh.transform_points(moved[2:],R)
    L=float(np.linalg.norm(np.diff(moved,axis=0),axis=1).sum())
    rows.append(dict(joint=joint,angle_rad=angle,route_length_m=L,delta_m=L-nom,scope='one joint separately, anchor split at point2 is explicit proxy; full serial attachment and swept Bowden route still unknown'))
  out.append(dict(id=r['id'],neutral_length_m=nom,sensitivities=rows,max_abs_delta_m=max(abs(x['delta_m']) for x in rows)))
 return out

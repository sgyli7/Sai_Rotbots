def hits(mats,posed):
 results=[]
 for p,m in mats:
  for old,n in posed:
   if np.any(m.bounds[0]>=n.bounds[1]) or np.any(n.bounds[0]>=m.bounds[1]):continue
   try:
    x=trimesh.boolean.intersection([m,n],engine='manifold');vol=float(x.volume) if len(x.faces) else 0.
    if vol>1e-10:results.append({'actuator_part':p['name'],'existing_C15_part':old['name'],'existing_role':old['role'],'existing_body':old['body'],'intersection_volume_m3':vol,'intersection_bounds_world_m':x.bounds.tolist()})
   except Exception as exc:results.append({'actuator_part':p['name'],'existing_C15_part':old['name'],'boolean_unresolved':str(exc)})
 return results

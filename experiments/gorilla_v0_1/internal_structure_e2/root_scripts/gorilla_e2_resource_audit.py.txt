def point(T,p):return (np.array(T)@np.r_[p,1])[:3]
def main():
 src=OUT/'scene.json';rp=OUT/'exports/internal_structure_e2_render_manifest.json';sp=OUT/'statics.json';scene,render,stat=[json.loads(p.read_text()) for p in (src,rp,sp)]

  assert err<3e-7,(part['name'],err)

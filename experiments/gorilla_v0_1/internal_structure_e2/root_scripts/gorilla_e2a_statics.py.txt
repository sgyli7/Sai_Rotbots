def lp(V,loads,models):
 force,moment=resultant(loads,np.zeros(3));E=np.vstack([np.ones(len(V)),V[:,0],V[:,1]]);target=np.array([-force[2],moment[1],-moment[0]]);E=np.column_stack([E,np.zeros(3)]);A=[];b=[]
 for m in models:
  k=np.array(m['coefficient_Nm_per_N']);c=m['bias_Nm'];pos=m['positive_capacity_reference_Nm'];neg=m['negative_capacity_reference_Nm']
  if min(pos,neg)<=0:raise ValueError('Nonpositive reference '+m['id'])
  A.extend([np.r_[k/pos,-1],np.r_[-k/neg,-1]]);b.extend([-c/pos,c/neg])
 A=np.asarray(A);b=np.asarray(b);cost=np.r_[np.zeros(len(V)),1.];res=linprog(cost,A_ub=A,b_ub=b,A_eq=E,b_eq=target,bounds=[(0,None)]*len(cost),method='highs')
 out={'solver_success':bool(res.success),'message':res.message,'physical_acceptance':False}
 if res.success:
  lam=np.array(res.ineqlin.marginals);y=np.array(res.eqlin.marginals);rc=cost-A.T@lam-E.T@y;dual=float(b@lam+target@y)
  err={'equality':float(abs(E@res.x-target).max()),'capacity':float(max(0,(A@res.x-b).max())),'variable_bound':float(max(0,-res.x.min())),'dual':float(max(0,lam.max(),-rc.min())),'gap':float(res.fun-dual)}
  valid=err['equality']<1e-5 and max(err['capacity'],err['variable_bound'],err['dual'])<1e-7 and abs(err['gap'])<1e-6
  if not valid:raise ValueError('LP certificate mismatch '+str(err))
  out.update(reference_minimax_peak_use=float(res.fun),conditional_reference_within_hypotheses=bool(res.fun<=1+1e-7),normal_reactions_N=res.x[:-1].tolist(),numerical_errors=err,dual_certificate={'inequality_multipliers':lam.tolist(),'equality_multipliers':y.tolist(),'nonnegative_reduced_costs':rc.tolist(),'objective_lower_bound':dual,'floating_tolerances_valid':valid},models_at_common_witness=[{**m,'demand_Nm_at_common_witness':float(m['bias_Nm']+np.array(m['coefficient_Nm_per_N'])@res.x[:-1])} for m in models])
 return out

def mapfn(origin,ranges,wh):

def face(p,k,x,axes,mapping,color):

def section(p,z,mapping,color):
 v=np.array(p['vertices_world_m']);f=np.array(p['faces'])
 for t in v[f]:
  dist=t[:,2]-z
  if dist.min()>1e-10 or dist.max()< -1e-10:continue
  pts=[]
  for k in range(3):
   a=t[k];b=t[(k+1)%3];da=dist[k];db=dist[(k+1)%3]
   if abs(da)<1e-10:pts.append(a[:2])
   if da*db<0:pts.append((a+(b-a)*(-da)/(db-da))[:2])
  if len(pts)>=2:dr.line([mapping(pts[0]),mapping(pts[1])],fill=color,width=2)

def tube_path(path,ro,ri):
 outs=[];ins=[]
 for aa,bb in zip(path[:-1],path[1:]):
  aa=np.array(aa);bb=np.array(bb);d=bb-aa;L=np.linalg.norm(d)
  if L<1e-8:continue
  z=d/L;ax=np.cross([0,0,1],z);an=np.linalg.norm(ax);angle=np.degrees(np.arccos(np.clip(z[2],-1,1)))
  o=mf.Manifold.cylinder(L,ro,ro,16);ii=mf.Manifold.cylinder(L+.0002,ri,ri,16).translate([0,0,-.0001])
  if an>1e-9:
   # quaternion-free Rodrigues matrix supported by transform
   a=ax/an;K=np.array([[0,-a[2],a[1]],[a[2],0,-a[0]],[-a[1],a[0],0]]);th=np.radians(angle);M=np.eye(3)+np.sin(th)*K+(1-np.cos(th))*(K@K);T=np.column_stack([M,aa]);o=o.transform(T);ii=ii.transform(T)
  elif z[2]<0:o=o.rotate([180,0,0]).translate(aa.tolist());ii=ii.rotate([180,0,0]).translate(aa.tolist())
  else:o=o.translate(aa.tolist());ii=ii.translate(aa.tolist())
  outs.append(o);ins.append(ii)

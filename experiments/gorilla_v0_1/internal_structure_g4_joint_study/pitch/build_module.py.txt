def box(center,size): return mf.Manifold.cube(size,center=True).translate(center)
def ring(center,ro,ri,length,axis=AY):
    s=mf.Manifold.cylinder(length,ro,ro,64,center=True)
    if ri: s=s-mf.Manifold.cylinder(length+.002,ri,ri,64,center=True)
    rot=tr.geometry.align_vectors([0,0,1],axis)
    return s.transform(rot[:3,:4]).translate(center)
def net(shapes):
    s=shapes[0]
    for t in shapes[1:]:s=s+t
    return s

    v,f=arrays(m);p={'name':'g4_'+name,'body':owner,'role':role,'vertices_world_m':v,'faces':f,
         'density_kg_m3':DENSITY,'material_assumption':'Conditional S355J2H7850kg/m3; no supplier/pressure/manufacturing qualification',
         'rgba':[.22,.30,.33,1],**extra};parts.append(p);return p
def frame(a,b,axis=AY):
    u=(b-a)/np.linalg.norm(b-a);x=np.array(axis)-u*(axis@u);x/=np.linalg.norm(x)
    return np.column_stack([x,np.cross(u,x),u])

        rows.append({'q_deg':float(q),'eye_length_m':float(np.linalg.norm(b-aa)),'J_m_per_rad':jj})
    lo=min(x['eye_length_m'] for x in rows);hi=max(x['eye_length_m'] for x in rows);stroke=hi-lo+.020
    return {'B_arm_x_m':bx,'fixed_anchor_xz_m':[-.080,.520],'moving_anchor_xz_m':[bx,.040],

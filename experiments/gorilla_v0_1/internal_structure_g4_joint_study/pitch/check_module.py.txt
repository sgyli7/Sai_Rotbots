def M(p):return mf.Manifold(mf.Mesh64(np.asarray(p['vertices_world_m'],np.float64,order='C'),np.asarray(p['faces'],np.uint64,order='C')))
def bounds_of(p):v=np.asarray(p['vertices_world_m']);return np.array([v.min(0),v.max(0)])
def overlap_box(a,b):return bool(np.all(a[1]>=b[0]) and np.all(b[1]>=a[0]))
def mp(p):
    t=tr.Trimesh(p['vertices_world_m'],p['faces'],process=False);u=t.mass_properties
    vol=float(u.volume)
    masses=p['mass_range_kg'] if p.get('density_kg_m3') is None else [vol*p['density_kg_m3']]*3
    return {'id':p['name'],'body':p['body'],'role':p['role'],'volume_m3':vol,'mass_range_kg':masses,
        'com_world_m':u.center_mass.tolist(),'inertia_at_com_world_kg_m2_range':[(u.inertia*x/vol).tolist() for x in masses],
        'mass_scope':'cataloguebearinguniformenvelopeinertia' if p['role']=='bearing_reference_envelope' else 'finiteconditionalsteelrho7850'}
def total(rows):
    out=[]
    for k in range(3):
        mass=sum(r['mass_range_kg'][k] for r in rows);com=sum(np.array(r['com_world_m'])*r['mass_range_kg'][k] for r in rows)/mass
        inertia=np.zeros((3,3))
        for r in rows:
            d=np.array(r['com_world_m'])-com;inertia+=np.array(r['inertia_at_com_world_kg_m2_range'][k])+r['mass_range_kg'][k]*(np.dot(d,d)*np.eye(3)-np.outer(d,d))
        out.append({'case':['low','nominal','high'][k],'mass_kg':mass,'com_world_m':com.tolist(),'inertia_at_com_world_kg_m2':inertia.tolist(),'inertia_eigenvalues':np.linalg.eigvalsh(inertia).tolist()})
    return out

            pairs.append({'a':a['name'],'b':b['name'],'a_body':a['body'],'b_body':b['body'],'volume_cm3':v*1e6,'classification':cls,'intended_interference_qualified':False})

            for mode in ['actuator_equal_share_bearings_take_roll_yaw','actuator_pair_takes_roll_bearings_take_yaw','neutral_mechanical_shoes_equal_bearings_take_roll_yaw']:

                        'local_journal_offset_m':.025,'local_journal_nominal_bending_MPa':radial*.025/Z/1e6,'does_not_take_complete_pitch_torque':True})

                            'ideal_required_pressure_MPa':abs(f)/area/1e6,'pressure_efficiency_screens':[{'pressure_MPa':p,'efficiency':e,'available_force_N':p*1e6*area*e,'satisfied':abs(f)<=p*1e6*area*e} for p in [21.,30.] for e in [1.,.8]],
                            'pin_double_shear_MPa':abs(f)/(2*math.pi*.04**2/4)/1e6,'pin_midspan76mm_bending_MPa':abs(f)*.076/4/(math.pi*.04**3/32)/1e6,
                            'eye_projected_bearing_MPa':abs(f)/(.04*.034)/1e6,'fork_projected_bearing_MPa':abs(f)/(2*.04*.014)/1e6})

                parentports += [{'point_world_m':r['point_world_m'],'force_on_parent_N':(-np.array(r['force_on_child_N'])).tolist()} for r in reactions]
                top_mount=C+[-.170,0.,.610]

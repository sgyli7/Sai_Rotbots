def rot(axis, q):
    u = np.array(axis, dtype=float)
    u /= np.linalg.norm(u)
    c, s = math.cos(q), math.sin(q)
    K = np.array([[0, -u[2], u[1]], [u[2], 0, -u[0]], [-u[1], u[0], 0]])
    return c * np.eye(3) + (1-c) * np.outer(u, u) + s * K

def cylinder_path(anchor, crank, axis, q_range_deg):
    a, b0 = np.array(anchor), np.array(crank)
    rows = []
    for q in np.linspace(*q_range_deg, 121):
        b = rot(axis, math.radians(q)) @ b0
        L = np.linalg.norm(b-a)
        u = (b-a)/L
        J = float(u @ np.cross(axis, b))
        rows.append({'q_deg': float(q), 'eye_length_m': float(L), 'J_m_per_rad': J,
                     'moving_eye_relative_pivot_m': b.tolist()})
    lengths = [r['eye_length_m'] for r in rows]
    js = [r['J_m_per_rad'] for r in rows]
    stroke = max(lengths)-min(lengths)+.020
    zero = min(lengths)-.010
    # G3 own construction provision, not a manufacturer cylinder drawing.
    minimum_zero = stroke + .176
    return {'anchor_relative_pivot_m': list(anchor), 'crank_relative_pivot_m': list(crank),
            'axis_parent_frame': list(axis), 'q_range_deg': q_range_deg, 'sample_count': 121,
            'eye_length_range_m': [min(lengths), max(lengths)],
            'signed_J_range_m_per_rad': [min(js), max(js)],
            'minimum_abs_J_m_per_rad': min(abs(j) for j in js),
            'stroke_with_10mm_each_end_reserve_m': stroke,
            'custom_zero_eye_m': zero, 'own_minimum_complete_zero_eye_m': minimum_zero,
            'own_zero_eye_margin_m': zero-minimum_zero,
            'own_zero_eye_rule_condition_met': zero >= minimum_zero,
            'manufacturer_drawing_qualified': False,
            'own_zero_rule_scope': '40mm eye pin, 38mm eye radius, own100mm caps/gland/overlap sum; not proof of seal, guidance, manufacture or stroke.'}

def seat_radial_reactions(axis, S, Fext, Mext):
    """Two radial seats separated by pressure-center S, no preload/axial force."""
    u = np.array(axis, dtype=float)
    if abs(float(u@Mext)) > 1e-6:
        raise ValueError('free-axis torque must first be supported by stated actuator/lock')
    # R1+R2=-F; S/2*u x(Rplus-Rminus)=-Mperp.
    delta = 2 * np.cross(u, Mext) / S
    Rplus, Rminus = (-Fext+delta)/2, (-Fext-delta)/2
    residual_F = Rplus+Rminus+Fext
    residual_M = np.cross(u*S/2,Rplus)+np.cross(-u*S/2,Rminus)+Mext
    radial = [float(np.linalg.norm(r-u*(u@r))) for r in [Rminus,Rplus]]
    return Rminus, Rplus, radial, float(np.abs(np.r_[residual_F,residual_M]).max())

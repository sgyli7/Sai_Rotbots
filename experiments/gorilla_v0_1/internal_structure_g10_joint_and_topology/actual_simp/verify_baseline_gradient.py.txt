def assemble_factor(B,D,V,dofs,factor,n):
    ke=np.einsum('eai,ab,ebj,e->eij',B,D,B,V*factor,optimize=True)
    rows=np.broadcast_to(dofs[:,:,None],ke.shape);cols=np.broadcast_to(dofs[:,None,:],ke.shape)
    return coo_matrix((ke.ravel(),(rows.ravel(),cols.ravel())),shape=(n,n)).tocsr()
def independent_variable_test():
    from skfem import MeshTet,Basis,ElementTetP1,ElementVector,BilinearForm,asm
    from skfem.helpers import sym_grad,ddot,trace
    n,t=p1.cube_mesh();K,V,B,D,df,_=p1.assemble(n,t);fac=np.linspace(.2,1.1,len(t));own=assemble_factor(B,D,V,df,fac,len(n)*3)
    basis=Basis(MeshTet(n.T,t.T,sort_t=False),ElementVector(ElementTetP1()),intorder=2)
    lam=E*NU/((1+NU)*(1-2*NU));mu=E/(2*(1+NU))
    @BilinearForm
    def form(u,v,w):return w.factor*(2*mu*ddot(sym_grad(u),sym_grad(v))+lam*trace(sym_grad(u))*trace(sym_grad(v)))
    independent=asm(form,basis,factor=np.repeat(fac[:,None],basis.X.shape[1],axis=1)).tocsr()
    rel=float(np.linalg.norm((own-independent).data)/np.linalg.norm(independent.data));rigid,_,_=p1.rigid_matrix(n)
    report={'actual_test':'variable positive element modulus numerical cube assembly against independent scikit-fem12.0.2','matrix_relative_error':rel,'six_rigid_modes_relative_error':float(np.linalg.norm(own@rigid)/(np.linalg.norm(own.data)*np.linalg.norm(rigid))),'pass':bool(rel<1e-11),'not_robot_mesh_or_physical_test':True}
    save('variable_assembly_test.json',report)
    if not report['pass']:raise ValueError('variable assembly test failed')
    event('variable_assembly_verified',relative_matrix_error=rel)
    return report
class Engine:
 def __init__(self,nodes,cells,F,coeff,C0,B,D,V,dofs):
    self.nodes=nodes;self.cells=cells;self.F=F;self.coeff=coeff;self.C0=C0;self.B=B;self.D=D;self.V=V;self.dofs=dofs
    self.rigid,_,_=p1.rigid_matrix(nodes);_,_,piv=qr(self.rigid.T,pivoting=True,mode='economic');self.fixed=piv[:6];assert np.linalg.matrix_rank(self.rigid[self.fixed])==6
    self.mask=np.ones(len(nodes)*3);self.mask[self.fixed]=0;self.P=diags(self.mask)
    lump=np.bincount(cells.ravel(),weights=np.repeat(V/4,4),minlength=len(nodes));self.w=np.repeat(lump/lump.sum(),3);self.gram=(self.rigid.T*self.w)@self.rigid
 def solve(self,factor,warm=None,label='state',maxseconds=420):
    begin=time.monotonic();K=assemble_factor(self.B,self.D,self.V,self.dofs,factor,len(self.nodes)*3)
    A=(self.P@K@self.P+diags((1-self.mask)*float(np.median(K.diagonal())))).tobsr(blocksize=(3,3));near=self.rigid.copy();near[self.fixed]=0
    event('amg_setup',label=label,nnz=int(K.nnz))
    ml=pyamg.smoothed_aggregation_solver(A,B=near,symmetry='symmetric',smooth=('jacobi',{'omega':4/3}),presmoother=('block_gauss_seidel',{'sweep':'symmetric'}),postsmoother=('block_gauss_seidel',{'sweep':'symmetric'}),max_coarse=100,coarse_solver='pinv')
    pre=ml.aspreconditioner(cycle='V');U=np.zeros_like(self.F);receipts=[]
    for j in range(self.F.shape[1]):
      f=self.F[:,j];count=[0];st=time.monotonic()
      if np.linalg.norm(self.rigid.T@f)/(np.linalg.norm(f)+1e-30)>1e-8:raise ValueError('literal F unbalanced, no projection')
      def cb(u):
       count[0]+=1
       if time.monotonic()-st>300 or time.monotonic()-begin>maxseconds:raise TimeoutError('bounded 8-basis state solve budget')
      x0=None if warm is None else warm[:,j]-self.rigid@np.linalg.solve(self.rigid[self.fixed],warm[self.fixed,j])
      u,info=cg(A,f*self.mask,x0=x0,M=pre,rtol=1e-9,atol=0,maxiter=1500,callback=cb)
      res=K@u-f;rel=float(np.linalg.norm(res)/np.linalg.norm(f));reaction=float(abs(res[self.fixed]).max())
      if info or rel>2e-7 or reaction>1e-3:raise ValueError(f'PCG failed {info} {rel} {reaction}')
      U[:,j]=u-self.rigid@np.linalg.solve(self.gram,(self.rigid.T*self.w)@u)
      rr={'basis_column':j,'iterations':count[0],'relative_full_K_residual':rel,'gauge_reaction_max_N':reaction,'seconds':time.monotonic()-st};receipts.append(rr);event('basis_complete',label=label,**rr)
    gram=self.F.T@U;C=np.einsum('iq,ij,jq->q',self.coeff,gram,self.coeff);rat=C/self.C0;phi=float(np.mean(rat**8)**(1/8))
    if not(np.isfinite(C).all() and (C>0).all()):raise ValueError('nonpositive compliance')
    strain=np.einsum('eai,eij->eaj',self.B,U[self.dofs]);weights=rat**7/(42*self.C0*phi**7);W=(self.coeff*weights)@self.coeff.T
    sensitivity_energy=np.einsum('eai,ab,ebj,ij,e->e',strain,self.D,strain,W,self.V,optimize=True)
    report={'label':label,'objective_p8':phi,'compliance_J_all42':C.tolist(),'relative_to_own_baseline_all42':rat.tolist(),'basis_receipts':receipts,'elapsed_seconds':time.monotonic()-begin,'literal_loads_projected':False,'gauge_is_not_physical_support':True}
    return U,sensitivity_energy,report

def build_filter(cent,V,design,fixed):
 tree=cKDTree(cent);ids=np.flatnonzero(design);mapidx=np.full(len(cent),-1,dtype=np.int32);mapidx[ids]=np.arange(len(ids));rows=[];cols=[];vals=[];fixedsum=np.zeros(len(ids));hs=np.zeros(len(ids));begin=time.monotonic()
 for st in range(0,len(ids),1024):
  ii=ids[st:st+1024];nb=tree.query_ball_point(cent[ii],RF,workers=1)
  rr=np.repeat(np.arange(st,st+len(ii),dtype=np.int32),[len(q) for q in nb]);jj=np.concatenate(nb).astype(np.int32);dist=np.linalg.norm(cent[jj]-cent[ids[rr]],axis=1);w=np.maximum(0,RF-dist)*V[jj];hs+=np.bincount(rr,weights=w,minlength=len(ids));ff=fixed[jj];fixedsum+=np.bincount(rr[ff],weights=w[ff],minlength=len(ids));ok=~ff;rows.append(rr[ok]);cols.append(mapidx[jj[ok]]);vals.append(w[ok])
  if st%30000<1024:event('filter_progress',rows=st,total=len(ids),seconds=time.monotonic()-begin)
 H=coo_matrix((np.concatenate(vals),(np.concatenate(rows),np.concatenate(cols))),shape=(len(ids),len(ids))).tocsr();save_npz(OUT/'density_filter_design_design.npz',H);np.savez_compressed(OUT/'filter_fixed_inventory.npz',design_element_ids=ids,denominator=hs,fixed_weight_sum=fixedsum)
 event('filter_complete',design_rows=len(ids),nonzeros=int(H.nnz),seconds=time.monotonic()-begin)
 return ids,H,hs,fixedsum

def boundary_binary(nodes,cells,fixed,rho,F):
 keep=(rho>.5)|fixed;kt=cells[keep];faces=np.vstack([kt[:,[0,2,1]],kt[:,[0,1,3]],kt[:,[0,3,2]],kt[:,[1,2,3]]]);ordered=np.sort(faces,axis=1);keys,inv,count=np.unique(ordered,axis=0,return_inverse=True,return_counts=True);boundary=faces[count[inv]==1]
 from scipy.sparse.csgraph import connected_components
 # Face-connected solids, including all islands (none silently removed).
 order=np.argsort(inv);facecell=np.tile(np.arange(len(kt)),4);ss=np.flatnonzero(count[inv[order]]==2);pair=ss[::2];aa=facecell[order[pair]];bb=facecell[order[pair+1]];graph=coo_matrix((np.ones(len(aa)),(aa,bb)),shape=(len(kt),len(kt))).tocsr();roots,labels=connected_components(graph,directed=False)
 edges=np.sort(np.vstack([boundary[:,[0,1]],boundary[:,[1,2]],boundary[:,[2,0]]]),axis=1);_,ec=np.unique(edges,axis=0,return_counts=True)
 active=np.unique(kt);mapping=np.full(len(nodes),-1,dtype=int);mapping[active]=np.arange(len(active));loaded=np.flatnonzero(np.any(F.reshape(len(nodes),3,-1)!=0,axis=(1,2)));loadedpresent=bool(np.all(mapping[loaded]>=0));bodyvol=abs(np.linalg.det(nodes[kt[:,1:]]-nodes[kt[:,0,None]]))/6
 np.savez_compressed(OUT/'binary_native_tet_domain.npz',points_m=nodes[active],tetra=mapping[kt],boundary_faces=mapping[boundary],original_kept_element_ids=np.flatnonzero(keep),original_kept_node_ids=active,fixed_original_element_ids=np.flatnonzero(fixed),component_labels=labels)
 report={'threshold_strict_rho_greater_than':.5,'fixed_elements_always_retained':bool(np.all(keep[fixed])),'kept_elements':len(kt),'kept_nodes':len(active),'actual_material_volume_m3':float(bodyvol.sum()),'actual_material_mass_kg':float(bodyvol.sum()*RHO),'face_connected_material_components':int(roots),'all_component_element_counts':np.bincount(labels).tolist(),'boundary_edges_not_degree2':int(np.count_nonzero(ec!=2)),'face_counts_above2':int(np.count_nonzero(count>2)),'loaded_nodes_retained':loadedpresent,'ersatz_in_binary':False,'islands_deleted_or_repaired':False,'CAD_or_manufacturing_qualified':False,'conditional_FE_admission':bool(roots==1 and np.all(ec==2) and np.all(count<=2) and loadedpresent)}
 save('binary_native_gate.json',report);return keep,active,mapping,report

"""Reassemble producer FEM fields with scikit-fem, including true void cells."""
from pathlib import Path
import hashlib
import json
import sys
import numpy as np
from scipy.sparse.linalg import splu
ROOT = next(p for p in Path(__file__).resolve().parents if (p / 'sai_robots_engineering_rules.md').exists())
sys.path.insert(0, str(ROOT / '.scratch/gorilla_topology_runtime'))
from skfem import MeshQuad, Basis, ElementQuad1, ElementVector, BilinearForm, asm
from skfem.helpers import sym_grad, ddot, div
import skfem
P = ROOT / '.scratch/gorilla_internal_g5_topology_benchmark'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
files = [P / name for name in ['input.json', 'mesh_loads.npz', 'density_fields.npz', 'solution_fields.npz', 'benchmark_report.json']]
before = {str(p.relative_to(ROOT)): sha(p) for p in files}
inp = json.loads(files[0].read_text()); report = json.loads(files[-1].read_text())
with np.load(P / 'mesh_loads.npz') as d:
    nodes = d['nodes_xz_m']; cells = d['q4_nodes_BL_BR_TR_TL']; F = d['load_Fx_Fz_dofs_by_case']; fixed = d['fixed_dofs']
with np.load(P / 'density_fields.npz') as d:
    rho = d['physical_density']; binary = d['threshold_binary']
with np.load(P / 'solution_fields.npz') as d:
    producer_fields = {k: d[k] for k in d.files}
E, nu = inp['E_Pa'], inp['nu']
lam, mu = E * nu / (1 - nu ** 2), E / (2 * (1 + nu))
@BilinearForm
def law(u, v, w):
    return w.factor * (2 * mu * ddot(sym_grad(u), sym_grad(v)) + lam * div(u) * div(v))

def verify(name, thickness, active, factor):
    ids = np.flatnonzero(active); used = np.unique(cells[ids]); map_cells = np.searchsorted(used, cells[ids])
    mesh = MeshQuad(nodes[used].T, map_cells.T)
    basis = Basis(mesh, ElementVector(ElementQuad1()), intorder=2)
    assert np.array_equal(basis.nodal_dofs, np.arange(len(used)*2).reshape(-1, 2).T)
    weights = np.repeat(factor[ids, None], basis.X.shape[-1], axis=1)
    K = asm(law, basis, factor=weights) * thickness
    old_dof = (2 * used[:, None] + [0, 1]).ravel(); force = F[old_dof]
    constrained = np.flatnonzero(np.isin(old_dof, fixed)); free = np.setdiff1d(np.arange(basis.N), constrained)
    U = np.zeros_like(force); U[free] = splu(K[free][:, free].tocsc()).solve(force[free])
    producer_U = producer_fields[name + '_U_dofs_cases'][old_dof]
    diff = float(np.max(np.abs(U - producer_U))); rel = diff / float(np.max(np.abs(producer_U)))
    compliances = [float(force[:, k] @ U[:, k]) for k in range(2)]
    sigma_fields = []
    for k in range(2):
        grad = basis.interpolate(U[:, k]).grad.mean(axis=-1)
        ex, ez = grad[0, 0], grad[1, 1]; gamma = grad[0, 1] + grad[1, 0]
        sx = E / (1 - nu*nu) * (ex + nu * ez) * factor[ids]
        sz = E / (1 - nu*nu) * (ez + nu * ex) * factor[ids]
        tau = mu * gamma * factor[ids]
        vm = np.sqrt(sx*sx - sx*sz + sz*sz + 3*tau*tau)
        sigma_fields.append(vm)
    vm = np.stack(sigma_fields, axis=1)
    producer_vm = producer_fields[name + '_center_VM_Pa'][ids]
    stress_rel = float(np.max(np.abs(vm - producer_vm)) / np.max(np.abs(producer_vm)))
    residual = K @ U - force
    return {'name': name, 'material_cell_count': len(ids), 'all_material_components_retained': True,
            'mass_kg': float(rho[ids].sum() if name == 'virtual' else len(ids)) * inp['geometry_m']['L'] * inp['geometry_m']['H'] / len(cells) * thickness * inp['density_kg_m3'],
            'mass_scope': 'Virtual density-weighted inventory, not actual manufactured material.' if name == 'virtual' else 'Homogeneous synthetic plate elements only.',
            'compliance_J': compliances, 'weighted_compliance_J': float(np.dot(inp['case_weights'], compliances)),
            'maximum_displacement_difference_m': diff, 'relative_field_difference': rel,
            'relative_center_VM_field_difference': stress_rel,
            'independent_free_residual_max_N': float(np.max(np.abs(residual[free]))),
            'passed_numerical_comparison': rel < 1e-7 and stress_rel < 1e-7}

ones = np.ones(len(cells)); all_cells = np.ones(len(cells), dtype=bool)
emin, p = inp['SIMP']['Emin_over_E'], inp['SIMP']['p']
results = [verify('full10', .010, all_cells, ones), verify('full5', .005, all_cells, ones),
           verify('virtual', .010, all_cells, emin + (1-emin)*rho**p),
           verify('binary', .010, binary, ones)]
assert before == {str(p.relative_to(ROOT)): sha(p) for p in files}
result = {'scope': 'Independent numerical verification of synthetic 2D tool only, not Gorilla structure performance.',
          'assembler': 'scikit-fem', 'version': skfem.__version__, 'input_hashes': before,
          'entry_sha256': sha(Path(__file__)), 'results': results,
          'all_numerical_comparisons_pass': all(r['passed_numerical_comparison'] for r in results),
          'optimization_converged': report['stopped_by_change_lt_p01'],
          'robot_weight_strength_or_durability_accepted': False}
Path(__file__).with_name('independent_field_comparison.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result, indent=2))

"""Independent signed tetrahedron volume/centroid and native pose inventory audit.
Does not import upstream builders/evaluators or use trimesh mass_properties.
"""
from pathlib import Path
from collections import Counter,defaultdict
import json,hashlib,datetime
import numpy as np
import trimesh as tm
ROOT=Path(__file__).resolve().parents[2]
OUT=Path(__file__).resolve().parent
ST=ROOT/'.scratch/gorilla_internal_f1_structure'
paths={k:ST/v for k,v in [('scene','structure_scene.json'),('interface','interface.json'),('mass_ledger','mass_ledger.json'),('checks','checks.json')]}
raw={k:p.read_bytes() for k,p in paths.items()}; d={k:json.loads(v) for k,v in raw.items()}
s=d['scene']; ledger=d['mass_ledger']; neutral={p['name']:p for p in s['parts']}

def triangles(p):
    fs=[]
    for f in p['faces']:
        for i in range(1,len(f)-1):fs.append([f[0],f[i],f[i+1]])
    return np.array(fs,dtype=int)
def tetra(p):
    vs=np.array(p['vertices_world_m'],float); fs=triangles(p); org=(vs.min(0)+vs.max(0))/2
    a,b,c=(vs-org)[fs].transpose(1,0,2)
    sv=np.einsum('ij,ij->i',a,np.cross(b,c))/6
    V=float(sv.sum()); center=(sv[:,None]*(a+b+c)/4).sum(0)/V+org
    edgecount=Counter(tuple(sorted((int(f[i]),int(f[(i+1)%3])))) for f in fs for i in range(3))
    return V,center,all(n==2 for n in edgecount.values())
def props(mesh):
    return tetra({'vertices_world_m':mesh.vertices.tolist(),'faces':mesh.faces.tolist()})
def mesh(p):return tm.Trimesh(vertices=p['vertices_world_m'],faces=triangles(p),process=False)
def union(ids,parts):
    ms=[mesh(parts[n]) for n in ids]
    return ms[0] if len(ms)==1 else tm.boolean.union(ms,engine='manifold')
def rigid_error(a,b):
    a=np.array(a);b=np.array(b); ac=a-a.mean(0);bc=b-b.mean(0)
    u,sv,vt=np.linalg.svd(ac.T@bc);rr=u@vt
    if np.linalg.det(rr)<0:u[:,-1]*=-1;rr=u@vt
    return float(np.max(np.linalg.norm(ac@rr-bc,axis=1))),float(np.linalg.det(rr))

errors=[]; eachpose=[]; base_props={n:tetra(p) for n,p in neutral.items()}
for pose in s['pose_scenes']:
    pm={p['name']:p for p in pose['parts']}; info={'pose':pose['id'],'part_count':len(pm),'same_ids':set(pm)==set(neutral),'rows':[],'max_rigid_vertex_residual_m':0.,'max_part_volume_delta_m3':0.}
    for n,bp in neutral.items():
        pp=pm[n];v,com,closed=tetra(pp);nv,nc,ncl=base_props[n];er,det=rigid_error(bp['vertices_world_m'],pp['vertices_world_m'])
        facesame=pp['faces']==bp['faces'];ownersame=(pp.get('body'),pp.get('mass_owner_body'),pp.get('material_union_group'))==(bp.get('body'),bp.get('mass_owner_body'),bp.get('material_union_group'))
        row={'id':n,'positive_volume_m3':v,'closed_edges':closed,'faces_same':facesame,'body_owner_group_same':ownersame,'volume_delta_m3':v-nv,'rigid_residual_m':er,'rotation_det':det}
        info['rows'].append(row);info['max_rigid_vertex_residual_m']=max(info['max_rigid_vertex_residual_m'],er);info['max_part_volume_delta_m3']=max(info['max_part_volume_delta_m3'],abs(v-nv))
        if not facesame or not ownersame or v<=0 or not closed or er>1e-8 or abs(v-nv)>1e-10:errors.append({'pose':pose['id'],**row})
    eachpose.append(info)

group_rows=[]; net_mass=0.; moment=np.zeros(3); unionmeshes={}
for g in ledger['material_groups']:
    ids=g['member_ids'];u=union(ids,neutral);V,C,closed=props(u);mass=V*g['density_kg_m3'];net_mass+=mass;moment+=mass*C;unionmeshes[g['group']]=u
    sv,sc,cl=tetra(neutral[ids[0]]) if len(ids)==1 else (None,None,None)
    declared_components=g.get('component_volumes_cm3',[])
    group_rows.append({'group':g['group'],'member_ids':ids,'body':g['body'],'mass_owner_body':g['mass_owner_body'],'union_signed_tetra_volume_m3':V,'union_centroid_world_m':C.tolist(),'density_kg_m3':g['density_kg_m3'],'mass_kg':mass,'closed_union':closed,'ledger_volume_delta_m3':V-g['net_volume_m3'],'ledger_centroid_delta_m':(C-np.array(g['centroid_world_m'])).tolist(),'ledger_mass_delta_kg':mass-g['mass_kg'],'declared_positive_surface_component_count':g.get('positive_connected_components'),'declared_component_volumes_sum_minus_net_cm3':sum(declared_components)-V*1e6,'component_count_scope':'surface connected components may include internal cavity boundaries; not solid-connectivity certification'})
# Every pose union is computed separately; volume invariant alone does not certify strength/attachment.
pose_union=[]
for pose in s['pose_scenes']:
    pm={p['name']:p for p in pose['parts']}; maxdelta=0.; details=[]
    for g,row in zip(ledger['material_groups'],group_rows):
        u=union(g['member_ids'],pm);V,C,closed=props(u);dv=V-row['union_signed_tetra_volume_m3'];maxdelta=max(maxdelta,abs(dv));details.append({'group':g['group'],'net_volume_m3':V,'centroid_world_m':C.tolist(),'volume_delta_vs_neutral_m3':dv,'closed_union':closed})
    pose_union.append({'pose':pose['id'],'max_union_volume_delta_m3':maxdelta,'groups':details})
materialparts=[p for p in s['parts'] if p.get('density_kg_m3',0)>0]; groupmembers=[n for g in ledger['material_groups'] for n in g['member_ids']]
refs=[p for p in s['parts'] if p.get('role') not in ['holding_mount_window'] and 'mass_range_kg' in p]
refsum=np.sum([p['mass_range_kg'] for p in refs],axis=0);candidate=refsum+net_mass
# Actuator geometry requires two finite closed material pieces, not eyes/pins/windows alone.
actaudit=[]
for a in d['interface']['actuators']:
    pieces=[p['name'] for p in s['parts'] if p.get('role')=='finite_actuator_material' and p.get('motion_interface_id')==a['id']]
    actaudit.append({'id':a['id'],'finite_material_piece_ids':pieces,'piece_count':len(pieces),'complete_barrel_rod_geometry_present':len(pieces)==2,'additional_tail_m':a['additional_tail_m'],'short_working_deficit_m':max(0,-a['additional_tail_m'])})
receipt={'schema':'gorilla_f1_independent_structure_audit_v1','audited_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'inputs':{k:{'path':str(paths[k]),'sha256':hashlib.sha256(raw[k]).hexdigest()} for k in paths},'method':'independent fan triangulation + signed tetrahedron volume/first moments; manifold Boolean union only for finite native assembly geometry; no upstream evaluator imported','native_pose_inventory':eachpose,'inventory_failures':errors,'native_material_groups':group_rows,'pose_union_invariance':pose_union,'mass_replay':{'density_material_part_count':len(materialparts),'material_group_count':len(group_rows),'material_member_id_occurrences':len(groupmembers),'material_members_unique':len(groupmembers)==len(set(groupmembers)),'material_parts_members_match':set(groupmembers)=={p['name'] for p in materialparts},'net_material_mass_kg':net_mass,'material_COM_world_m':(moment/net_mass).tolist(),'source_net_material_mass_kg':ledger['net_material_mass_kg'],'net_material_delta_kg':net_mass-ledger['net_material_mass_kg'],'reference_count':len(refs),'reference_sum_range_kg':refsum.tolist(),'reference_delta_range_kg':(refsum-np.array(ledger['component_reference_mass_range_kg'])).tolist(),'candidate_structure_scope_range_kg':candidate.tolist(),'not_whole_robot_mass':True,'tensor_qualified':False},'actuator_inventory':actaudit,'full_finite_cylinders':sum(r['complete_barrel_rod_geometry_present'] for r in actaudit),'missing_finite_cylinders':[r['id'] for r in actaudit if not r['complete_barrel_rod_geometry_present']],'interfaces_not_physics_accepted':True,'system_final_audit_pending':True}
(OUT/'structure_receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps({'mass_replay':receipt['mass_replay'],'inventory_failures':errors,'max_pose_volume_m3':max(x['max_union_volume_delta_m3'] for x in pose_union),'full_finite_cylinders':receipt['full_finite_cylinders'],'missing':receipt['missing_finite_cylinders'],'max_group_volume_delta':max(abs(g['ledger_volume_delta_m3']) for g in group_rows),'max_group_centroid_delta_m':max(np.max(np.abs(g['ledger_centroid_delta_m'])) for g in group_rows),'component_volume_note_groups':[g['group'] for g in group_rows if abs(g['declared_component_volumes_sum_minus_net_cm3'])>.01]},indent=2))

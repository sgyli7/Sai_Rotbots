"""Independently recompute source-bound conditional gravity, never fill unknown mass."""
from pathlib import Path
import hashlib,json
import numpy as np

HERE=Path(__file__).resolve().parent;S=HERE.parent
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
path=S/'gorilla_internal_g9_upper_load_scope/upper_load_scope.json';src=json.loads(path.read_text())
all_rows=src['inventory_rows']+src['structure_rows']+src['complete_reference_rows']+[src['new_A_material_row']]
by_id={r['id']:r for r in all_rows};ids=src['conditional_diagnostic_subset_ids']
assert len(by_id)==len(all_rows) and len(set(ids))==len(ids)
rows=[by_id[x] for x in ids];g=src['conditional_subset_gravity']['g_m_s2']
pivot=np.asarray(src['historical_roll_body_tree_pivot_m'],float)
def subtotal(items):
    mass=np.asarray([r['mass_range_kg'] for r in items],float)
    com=np.asarray([r['source_COM_world_proxy_m'] for r in items],float)
    assert mass.shape==(len(items),3) and com.shape==(len(items),3)
    F=np.zeros((len(items),3));F[:,2]=-mass[:,1]*g
    W=np.c_[F,np.cross(com-pivot,F)]
    wc=com-pivot
    coeff=np.c_[np.zeros((len(items),2)),np.full(len(items),-g),-g*wc[:,1],g*wc[:,0],np.zeros(len(items))]
    ends=coeff[:,:,None]*mass[:,None,[0,2]]
    return dict(row_count=len(items),mass_range_kg=mass.sum(0).tolist(),
                nominal_proxy_COM_world_m=(mass[:,1,None]*com).sum(0).tolist(),
                nominal_wrench_Fxyz_Mxyz=W.sum(0).tolist(),
                independent_component_wrench_bounds=np.stack([ends.min(2).sum(0),ends.max(2).sum(0)],1).tolist())

A=subtotal(rows);A['nominal_proxy_COM_world_m']=(np.asarray(A['nominal_proxy_COM_world_m'])/A['mass_range_kg'][1]).tolist()
old=src['conditional_subset_gravity']
mass_delta=np.asarray(A['mass_range_kg'])-old['mass_range_sum_kg']
w_delta=np.asarray(A['nominal_wrench_Fxyz_Mxyz'])-old['nominal_gravity_Fxyz_Mxyz_at_roll_reference']
assert np.max(abs(mass_delta))<1e-9 and np.max(abs(w_delta))<1e-8
B_rows=[src['new_B_material_row'] if r['id']==src['new_A_material_row']['id'] else r for r in rows]
B=subtotal(B_rows);B['nominal_proxy_COM_world_m']=(np.asarray(B['nominal_proxy_COM_world_m'])/B['mass_range_kg'][1]).tolist()
layouts={}
for name in ['layout_1','layout_2']:
    mp=S/'gorilla_internal_g9_module_placement'/name/'moved_group_mass_positions.json'
    moved=json.loads(mp.read_text());groups=moved['complete_moved_groups']
    member_group={member:group for group in groups for member in group['complete_member_ids']}
    shifted=[];changed=[];not_upper=[];ambiguous=[]
    for original in B_rows:
        r=dict(original);members=r.get('native_member_ids',[])
        hits=[member_group[m] for m in members if m in member_group]
        if hits:
            names={q['id'] for q in hits}
            if len(names)!=1 or len(hits)!=len(members):
                ambiguous.append(r['id']);continue
            group=hits[0];matrix=np.asarray(group['incremental_rigid_transform_world'],float)
            r['source_COM_world_proxy_m']=(matrix@np.r_[r['source_COM_world_proxy_m'],1.])[:3].tolist()
            r['gravity_owner_candidate']=group['candidate_gravity_owner']
            changed.append(r['id'])
            if r['gravity_owner_candidate'] not in src['candidate_upper_body_names']:
                not_upper.append(r['id']);continue
        shifted.append(r)
    result=subtotal(shifted);result['nominal_proxy_COM_world_m']=(np.asarray(result['nominal_proxy_COM_world_m'])/result['mass_range_kg'][1]).tolist()
    layouts[name]={'scene_sha256':moved['scene_stored_sha256'],'movement_metadata_sha256':sha(mp),
                   'changed_row_ids':changed,'conditional_owner_changed_outside_upper_ids':not_upper,
                   'ambiguous_rows_not_filled':ambiguous,'conditional_same_selected_row_subtotal':result,
                   'module_installation_or_routing_accepted':False}
report={'robot_id':'gorilla_v0_1','producer_sha256':sha(__file__),'source_inventory_sha256':sha(path),
        'scope':'633 selected dry inventory rows only, with unqualified COM proxies and conditional steel. Independent arithmetic and motion sensitivity, not upper/whole mass or load contract.',
        'baseline_A_independent_sum':A,'baseline_B_independent_sum':B,
        'A_mass_max_abs_difference_kg':float(abs(mass_delta).max()),
        'A_wrench_max_abs_difference_N_or_Nm':float(abs(w_delta).max()),'two_actual_moved_layout_sensitivities':layouts,
        'gravity_formula':'F=(0,0,-m*g); M=(COM-pivot) cross F; six-component bounds from each row endpoints.',
        'whole_mass_COM_I_or_actual_support_capacity_accepted':False,
        'missing_fluid_output_split_adapter_inventory_or_external_task_load_filled':False,
        'physical_unknowns_or_absent_parts_treated_as_zero':False}
(HERE/'upper_subset_arithmetic_review.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'A':A['mass_range_kg'],'B':B['mass_range_kg'],'A_wrench_delta':report['A_wrench_max_abs_difference_N_or_Nm'],
                  'moved_layouts':{k:v['conditional_same_selected_row_subtotal'] for k,v in layouts.items()}},indent=2))

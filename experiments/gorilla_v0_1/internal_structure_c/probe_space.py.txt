def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def skin_surface(p,which):
 v=np.asarray(p['vertices_world_m']);n=len(v)//2
 faces=[f for f in p['faces'] if (max(f)<n if which=='outer' else min(f)>=n)]
 f=[[q[0],q[i],q[i+1]] for q in faces for i in range(1,len(q)-1)]
 return trimesh.Trimesh(v,f,process=False)
def ray_hits(surfaces,origin,direction,max_distance=.8):
 hits=[];d=np.asarray(direction)
 for name,t in surfaces:
  if not len(t):continue
  a=t[:,0];e1=t[:,1]-a;e2=t[:,2]-a
  h=np.cross(d,e2);det=np.einsum('ij,ij->i',e1,h);ok=abs(det)>1e-11;inv=np.zeros(len(det));inv[ok]=1/det[ok]
  q=np.asarray(origin)-a;u=inv*np.einsum('ij,ij->i',q,h);c=np.cross(q,e1);v=inv*np.einsum('ij,j->i',c,d);dist=inv*np.einsum('ij,ij->i',e2,c)
  m=ok&(u>=-1e-8)&(v>=-1e-8)&(u+v<=1+1e-8)&(dist>1e-7)&(dist<=max_distance)
  for value in dist[m]:hits.append((float(value),name))
 hits.sort();return hits[0] if hits else None

def cylinder(center,axis,diameter,length,clearance):
 # Circumscribed 128gon conservatively contains the true outer circle.
 radius=(diameter/2+clearance)/math.cos(math.pi/128)
 m=trimesh.creation.cylinder(radius=radius,height=length+2*clearance,sections=128)
 m.apply_transform(trimesh.geometry.align_vectors([0,0,1],axis));m.apply_translation(center);return m

def collisions(probe,parts,cache,threshold=1e-10):
 lo,hi=probe.bounds;out=[]
 for p in parts:
  m=cache[p['name']]
  if np.any(m.bounds[1]<lo-1e-8) or np.any(m.bounds[0]>hi+1e-8):continue
  try:
   hit=trimesh.boolean.intersection([probe,m],engine='manifold');volume=float(hit.volume)
   if volume>threshold:out.append({'part':p['name'],'body':p['body'],'role':p['role'],'volume_m3':volume,
              'intersection_bounds_world_m':hit.bounds.tolist(),'status':'positive_material_or_envelope_overlap'})
  except Exception as exc:out.append({'part':p['name'],'status':'boolean_unresolved','error':str(exc)})
 return out

def pose_parts(parts,site,angle,pivot):
 t=math.radians(angle);R=np.array([[math.cos(t),0,math.sin(t)],[0,1,0],[-math.sin(t),0,math.cos(t)]]);result=[]
 moving=['left_upper_arm','left_forearm','left_palm','left_finger','left_thumb'] if site=='shoulder' else ['left_middle_shank','left_distal_shank','left_foot','left_forefoot_display','left_heel_display']
 for p in parts:
  q=p.copy()
  if any(p['body'].startswith(x) for x in moving):q['vertices_world_m']=((np.asarray(p['vertices_world_m'])-pivot)@R.T+pivot).tolist()
  result.append(q)
 return result

def radial_limits(armor,pivot,length,clearance):
 surfaces=[(p['name'],skin_surface(p,'inner').triangles) for p in armor if p.get('cover_construction')=='outer_skin_inner_skin_boundary_returns_open_cavity']
 rows=[]
 for offset in (-length/2,0,length/2):
  origin=np.asarray(pivot)+[0,offset,0];directions=[]
  for theta in np.arange(32)*2*math.pi/32:
   hit=ray_hits(surfaces,origin,[math.cos(theta),0,math.sin(theta)],.45)
   directions.append({'angle_rad':float(theta),'nearest_inner_radius_m':None if hit is None else hit[0],
                      'limiting_part':None if hit is None else hit[1]})
  values=[v['nearest_inner_radius_m'] for v in directions if v['nearest_inner_radius_m'] is not None]
  rows.append({'axial_offset_Y_m':offset,'directions':directions,'finite_ray_count':len(values),
      'minimum_observed_inner_radius_m':min(values) if values else None,
      'conditional_diameter_from_finite_min_ray_minus_clearance_m':None if not values else max(0,2*(min(values)-clearance)),
      'complete_enclosure_proved':False})
 return rows

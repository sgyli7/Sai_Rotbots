def digest(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(2**20), b''):
            h.update(block)
    return h.hexdigest()

def compare(actual, claimed, label):

    e = abs(actual - claimed)
    max_error = max(max_error, e)
    comparisons.append({'field': label, 'actual': actual, 'claimed': claimed, 'abs_error': e})
    if e > 1e-10 * max(1., abs(actual)):
        errors.append(comparisons[-1])

"""Source-preserving OCC volume meshing with actual CAD port physical groups."""
from pathlib import Path
import argparse,hashlib,json,time,itertools
import numpy as np
from scipy.sparse import coo_matrix
from scipy.sparse.csgraph import connected_components
import gmsh
from cad_surface_helpers import C,get_surface_samples

O=Path(__file__).resolve().parent;sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
ap=argparse.ArgumentParser();ap.add_argument('--variant',choices=['a','b'],required=True);ap.add_argument('--level',choices=['coarse'],default='coarse');a=ap.parse_args();O=O/a.variant
D=O/a.level;D.mkdir(exist_ok=False);start=time.time()
gmsh.initialize([],False,False);gmsh.option.setNumber('General.Terminal',0);gmsh.option.setNumber('General.NumThreads',4)
gmsh.model.add('gorilla_g8_carrier_'+a.variant+'_'+a.level);vv=gmsh.model.occ.importShapes(str(O/'carrier_exact.brep'))
gmsh.model.occ.synchronize();solids=gmsh.model.getEntities(3);assert len(solids)==1,solids
groups={i:[] for i in range(1,8)};names={1:'journal_radial_negative_y',2:'journal_radial_positive_y',3:'moving_pin_bore_negative_y',4:'moving_pin_bore_positive_y',5:'stop_contact_negative_y',6:'stop_contact_positive_y',7:'downstream_mount'}
bd=gmsh.model.getBoundary(solids,combined=True,oriented=False,recursive=False)
for dim,f in bd:
    typ=gmsh.model.getType(dim,f);local=get_surface_samples(f)-C
    if typ=='Cylinder':
        rj=np.hypot(local[:,0],local[:,2]);rp=np.hypot(local[:,0]-.320,local[:,2]-.040)
        if np.max(abs(rj-.02995))<1e-7:groups[1 if local[:,1].mean()<0 else 2].append(f)
        elif np.max(abs(rp-.02020))<1e-7:groups[3 if local[:,1].mean()<0 else 4].append(f)
    if typ=='Plane' and np.max(abs(local[:,2]-.088))<1e-7:
        com=np.asarray(gmsh.model.occ.getCenterOfMass(2,f))-C
        if .119<=com[0]<=.181 and abs(abs(com[1])-.075)<.024:groups[5 if com[1]<0 else 6].append(f)
    if typ=='Plane' and np.max(abs(local[:,0]-.350))<1e-7:
        com=np.asarray(gmsh.model.occ.getCenterOfMass(2,f))-C
        if abs(com[1])<=.12001 and -.09001<=com[2]<=.06001:groups[7].append(f)
assert all(groups.values()),groups
for i,ff in groups.items():gmsh.model.addPhysicalGroup(2,ff,i);gmsh.model.setPhysicalName(2,i,names[i])
gmsh.model.addPhysicalGroup(3,[solids[0][1]],101);gmsh.model.setPhysicalName(3,101,'finite_carrier_material')
coarse=a.level=='coarse';bulk=.006 if coarse else .0045;port=.0025 if coarse else .0016
gmsh.option.setNumber('Mesh.MeshSizeMin',.0006 if coarse else .00045)
gmsh.option.setNumber('Mesh.MeshSizeMax',bulk);gmsh.option.setNumber('Mesh.MeshSizeFromCurvature',24 if coarse else 32)
gmsh.option.setNumber('Mesh.Algorithm3D',10);gmsh.option.setNumber('Mesh.Optimize',1);gmsh.option.setNumber('Mesh.OptimizeThreshold',.30)
dist=gmsh.model.mesh.field.add('Distance');gmsh.model.mesh.field.setNumbers(dist,'SurfacesList',sum(groups.values(),[]));gmsh.model.mesh.field.setNumber(dist,'Sampling',36)
threshold=gmsh.model.mesh.field.add('Threshold')
for key,val in [('InField',dist),('SizeMin',port),('SizeMax',bulk),('DistMin',.005),('DistMax',.035)]:gmsh.model.mesh.field.setNumber(threshold,key,val)
gmsh.model.mesh.field.setAsBackgroundMesh(threshold)
print('PHASE',a.level,'CAD_LOADED','ports',[len(groups[i]) for i in range(1,8)],'sizes',bulk,port,flush=True)
gmsh.model.mesh.generate(3)
print('PHASE',a.level,'MESH_GENERATED',round(time.time()-start,2),flush=True)
types,element_tags,connectivity=gmsh.model.mesh.getElements(3);assert list(types)==[4],types
node_tags,xyz,_=gmsh.model.mesh.getNodes();points=np.asarray(xyz).reshape(-1,3)
order=np.argsort(node_tags);tags=np.asarray(node_tags)[order];points=points[order]
tetra=np.searchsorted(tags,np.asarray(connectivity[0]).reshape(-1,4)).astype(np.int64)
p=points[tetra];signed=np.einsum('ij,ij->i',np.cross(p[:,1]-p[:,0],p[:,2]-p[:,0]),p[:,3]-p[:,0])/6
assert np.all(signed>0),'Nonpositive tetrahedra rejected, not reoriented'
# Exact outward face rule for positively oriented tets, not source normal repair.
all_faces=np.concatenate([tetra[:,idx] for idx in [[0,2,1],[0,1,3],[0,3,2],[1,2,3]]])
canonical=np.sort(all_faces,axis=1);uniq,first,count=np.unique(canonical,axis=0,return_index=True,return_counts=True)
assert np.all(count<=2),'Non-manifold tetra face adjacency'
boundary=all_faces[first[count==1]];canonical_boundary=np.sort(boundary,axis=1)
# CAD entity and physical IDs mapped from original surface element connectivity.
lookup={tuple(row):i for i,row in enumerate(canonical_boundary)}
physical=np.zeros(len(boundary),np.int32);entity=np.zeros(len(boundary),np.int32);matched=np.zeros(len(boundary),np.int32)
face2group={f:i for i,ff in groups.items() for f in ff}
for dim,f in bd:
    typ,et,en=gmsh.model.mesh.getElements(2,f);assert list(typ)==[2],typ
    triangles=np.searchsorted(tags,np.asarray(en[0]).reshape(-1,3))
    for row in np.sort(triangles,axis=1):
        k=lookup.get(tuple(row));assert k is not None,'CAD triangle missing from tet boundary'
        matched[k]+=1;entity[k]=f;physical[k]=face2group.get(f,0)
assert np.all(matched==1),'Boundary-to-CAD identity mismatch'
np.savez_compressed(D/'carrier_tetra.npz',points_m=points,tetra=tetra,boundary_faces=boundary,
    boundary_physical_tags=physical,boundary_cad_surface_tags=entity,node_tags_original=tags,
    tetra_element_tags_original=np.asarray(element_tags[0]))
np.savez_compressed(D/'carrier_boundary.npz',vertices_world_m=points,faces=boundary,
    boundary_physical_tags=physical,boundary_cad_surface_tags=entity)
gmsh.write(str(D/'carrier_tetra.msh'))
port_rows=[]
for i in range(1,8):
    ff=groups[i];samples=np.concatenate([get_surface_samples(f) for f in ff])
    analytic={'axis_world':[0,1,0],'centerline_xz_world_m':[-.06,.30],'radius_m':.02995} if i in [1,2] else {
        'axis_world':[0,1,0],'centerline_xz_world_m':[.26,.34],'radius_m':.02020} if i in [3,4] else {
        'plane_axis_world':[0,0,1],'plane_value_world_m':.388} if i in [5,6] else {'plane_axis_world':[1,0,0],'plane_value_world_m':.290}
    port_rows.append({'physical_tag':i,'name':names[i],'cad_surface_tags':ff,
        'cad_area_m2':sum(gmsh.model.occ.getMass(2,f) for f in ff),'analytic_geometry':analytic,
        'surface_bounds_world_m':[gmsh.model.getBoundingBox(2,f) for f in ff],
        'boundary_triangle_count':int(np.sum(physical==i)),
        'load_distribution_actual_bearing_contact_or_pin_fit_qualified':False})
qualities=gmsh.model.mesh.getElementQualities(element_tags[0],'minSICN');gmsh.finalize()
vol=signed;length2=np.stack([np.sum((p[:,u]-p[:,v])**2,axis=1) for u,v in itertools.combinations(range(4),2)],axis=1)
mean_ratio=12*np.cbrt((3*vol)**2)/length2.sum(1)
edges=np.concatenate([tetra[:,[u,v]] for u,v in itertools.combinations(range(4),2)])
graph=coo_matrix((np.ones(len(edges)),(edges[:,0],edges[:,1])),shape=(len(points),len(points))).tocsr();roots,_=connected_components(graph,directed=False)
cad=json.loads((O/'cad_source_manifest.json').read_text());bv=points[boundary]
surface_volume=float(np.einsum('ij,ij->i',bv[:,0],np.cross(bv[:,1],bv[:,2])).sum()/6)
receipt={'robot_id':'gorilla_v0_1','level':a.level,'producer_sha256':sha(Path(__file__)),
    'cad_brep_sha256':sha(O/'carrier_exact.brep'),'cad_source_manifest_sha256':sha(O/'cad_source_manifest.json'),
    'mesh_sha256':sha(D/'carrier_tetra.npz'),'boundary_mesh_sha256':sha(D/'carrier_boundary.npz'),
    'runtime_manifest_sha256':sha(O/'tool_runtime_manifest.json'),'world_SI':True,'root_shift_z_m':0,
    'points':len(points),'tetra':len(tetra),'boundary_triangles':len(boundary),'material_node_graph_roots':int(roots),
    'unused_nodes':len(points)-len(np.unique(tetra)), 'negative_or_zero_tetra_count':int(np.sum(signed<=0)),
    'minimum_tetra_volume_m3':float(signed.min()),'maximum_tetra_volume_m3':float(signed.max()),
    'tetra_volume_m3':float(signed.sum()),'boundary_signed_volume_m3':surface_volume,'exact_CAD_volume_m3':cad['actual_volume_m3'],
    'linear_mesh_volume_relative_error_vs_CAD':float(signed.sum()/cad['actual_volume_m3']-1),
    'material_tet_vs_oriented_boundary_volume_error_m3':float(surface_volume-signed.sum()),
    'mean_ratio_quality_quantiles':np.quantile(mean_ratio,[0,.01,.05,.5,.95,1]).tolist(),
    'minSICN_quantiles':np.quantile(qualities,[0,.01,.05,.5,.95,1]).tolist(),
    'mean_ratio_below_0p01':int(np.sum(mean_ratio<.01)),'tetra_below_guard_1e_20':int(np.sum(signed<=1e-20)),
    'mesh_bulk_size_m':bulk,'mesh_port_size_m':port,'geometry_changed_by_meshing':False,
    'linear_chord_geometry_vs_exact_curved_CAD_difference_reported':True,
    'all_CAD_boundary_triangles_exactly_matched_tet_boundary':True,
    'boundary_orientation':'Outward positive-tetrahedron induced face order; original surface triangles mapped by undirected node triple, no native normal repair',
    'source_boundary_face_tags_not_G4_facets':True,'physical_release':False,'FE_loads_or_optimization_not_run':True,
    'mesh_quality_conditional_baseline_pass':bool(roots==1 and signed.min()>1e-20 and mean_ratio.min()>=.01 and abs(signed.sum()/cad['actual_volume_m3']-1)<.01),
    'uniform_two_layer_through6mm_wall_proof':False,'elapsed_s':time.time()-start}
(D/'mesh_receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
(D/'port_groups.json').write_text(json.dumps({'robot_id':'gorilla_v0_1','cad_brep_sha256':sha(O/'carrier_exact.brep'),
    'cad_source_manifest_sha256':sha(O/'cad_source_manifest.json'),'mesh_sha256':sha(D/'carrier_tetra.npz'),
    'boundary_mesh_sha256':sha(D/'carrier_boundary.npz'),'groups':port_rows,'world_SI':True,'original_facet_markers_reused':False},indent=2)+'\n')
print('RECEIPT',a.level,len(points),len(tetra),'quality',mean_ratio.min(),'volume_error',receipt['linear_mesh_volume_relative_error_vs_CAD'],'sha',receipt['mesh_sha256'],flush=True)

"""Restore selected exact G8 records; full42 loads are included; large fields stay separate."""
import argparse, gzip, hashlib, json
from pathlib import Path
parser = argparse.ArgumentParser()
parser.add_argument('--root', type=Path, required=True)
root = parser.parse_args().root.resolve()
here = Path(__file__).resolve().parent
data = json.loads((here/'snapshot_manifest.json').read_text())
for row in data['files']:
    source = (here/row['stored_path']).resolve()
    if not source.is_relative_to(here):
        raise ValueError('Source escapes snapshot')
    stored = source.read_bytes()
    assert hashlib.sha256(stored).hexdigest() == row['stored_sha256']
    raw = gzip.decompress(stored) if row['encoding']=='gzip' else stored
    assert hashlib.sha256(raw).hexdigest() == row['original_sha256']
    target = (root/row['original_path']).resolve()
    if not target.is_relative_to(root/'.scratch'):
        raise ValueError('Restore escapes scratch')
    if target.exists() and target.read_bytes()!=raw:
        raise ValueError('Refuse overwriting changed file: '+str(target))
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(raw)
print('Restored', len(data['files']), 'selected exact G8 records; no runtime or large fields.')

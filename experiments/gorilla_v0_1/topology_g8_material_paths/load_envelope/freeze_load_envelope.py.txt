"""Finalize a completed, audited envelope snapshot; never changes upstream files."""
from pathlib import Path
import hashlib,json
OUT=Path(__file__).resolve().parent;ROOT=OUT.parents[1]
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def save(p,d):Path(p).write_text(json.dumps(d,indent=2,ensure_ascii=False,allow_nan=False)+'\n')

def main():
    ref=json.loads((OUT/'full_load_reference.json').read_text());datasets=[];inputs=set()
    for report_path in sorted(OUT.glob('*_report.json')):
        r=json.loads(report_path.read_text());prefix=report_path.name[:-len('_report.json')];receipt_path=OUT/(prefix+'_verification.json')
        if not receipt_path.exists():raise ValueError('completed dataset lacks independent audit: '+prefix)
        v=json.loads(receipt_path.read_text())
        if not v['all_gates_pass'] or v['hashes']['report']!=sha(report_path):raise ValueError('independent audit mismatch')
        if r['mapper_sha256']!=sha(OUT/'assemble_load_envelope.py'):raise ValueError('producer script changed after run')
        if v['hashes']['independent_verifier']!=sha(OUT/'verify_load_envelope.py'):raise ValueError('verifier changed after run')
        for key,hashkey in [('mesh_path','mesh_sha256'),('CAD_source_path','CAD_source_sha256'),('physicalgroup_report_path','physicalgroup_report_sha256')]:
            p=Path(r[key]);inputs.add(p)
            if sha(p)!=r[hashkey]:raise ValueError('actual source changed after run')
        meshdir=Path(r['mesh_path']).parent;caddir=Path(r['CAD_source_path']).parent
        for p in [meshdir/'mesh_receipt.json',caddir/'cad_source_manifest.json',caddir/'carrier_exact.step',caddir/'freeze_manifest.json']:
            if p.exists():inputs.add(p)
        datasets.append((prefix,r,v))
    for p in [ROOT/'.scratch/gorilla_internal_g4_pitch_module/module_scene.json',ROOT/'.scratch/gorilla_internal_g4_pitch_module/optimization_handoff.json',ROOT/'.scratch/gorilla_internal_g4_pitch_module/report.json',ROOT/'.scratch/gorilla_internal_g7_load_ports/assemble_g7_load_ports.py',ROOT/'.scratch/gorilla_internal_g7_load_ports/interface_contract.json',ROOT/'.scratch/gorilla_internal_g6_load_ports/assemble_load_ports.py']:inputs.add(p)
    table=[]
    for prefix,r,v in datasets:
        table.append(f"| {prefix} | {r['actual_mesh_shape'][0]} | 42 | {r['SVD_numerical_rank']} | {v['max_port_force_error_N']:.4g} | {v['max_port_moment_error_Nm']:.4g} | {v['basis_reconstruction_max_abs_N']:.4g} |")
    text=f'''# G8 full conditional load envelope

The complete original G4 42-case static external-load domain is preserved, using literal point locations and six-dimensional wrenches from the frozen handoff. It contains 21 factor1 and 21 factor1.5 cases; ground force is respectively 29430 and 44145 N. This is a 3t static compression reference and its 1.5 sensitivity, with no robot/module gravity, walking, dynamic payload or real contact qualification.

There are 12 cases at −25°, 18 at neutral and 12 at +25°, and 14 cases each at original-world CoP [0.25,0.55,0], [0.25,0.65,0] and [0.25,0.75,0] m. Allocation is 18 equal actuator / 18 differential actuator / 6 neutral static-stop cases. No pose, CoP, factor or allocation is omitted. All case indices and basis indices are zero-based original G4 array indices.

The child geometry is evaluated in its neutral material reference frame. Each source point is mapped as pivot+Ry(q)ᵀ(x−pivot), and both force and the local couple rotate by Ry(q)ᵀ. This is a rigid change of coordinates; literal force, point and couple are retained with an inverse-transform check, and the load domain remains complete.

Actual CAD physical groups 1–7 select each new loading patch. No old facet IDs or nearest-pivot selection are used. Source CAD, mesh and group-report hashes are checked before mapping. The G7 expected-coordinate contract is independently checked against actual new surface nodes/radii/planes/outward normals. Each group's area-lumped arbitrary vector traction receives a full-wrench-preserving correction, with no global equilibrium projection. Generic rank6 traction can contain axial, tangential and tensile components, so these are conditional numerical load inputs, not pin/bearing/stop contact pressures. G7 case18 compression-only radial pressure remains a separate rank4 conditional probe, not a solved real contact state.

| Dataset | Nodes | Original cases | Direct SVD numerical rank | Max independent port F error N | Max independent port M error Nm | Max basis reconstruction error N |
|---|---:|---:|---:|---:|---:|---:|
'''+ '\n'.join(table)+'''

Each load NPZ contains one F_xyz_N_cases[node,xyz,42] array, the chosen actual original case columns basis_F_xyz_N_cases, original_case_coefficients[rank,42], basis_case_indices, loaded_node_ids and the complete direct thin-SVD singular spectrum. Mesh arrays are referenced once by their immutable source path/hash and are not repeated for each case. For one unchanged linear stiffness and gauge system, solving the actual basis columns then multiplying their solutions by original_case_coefficients reproduces all original load solutions to the recorded load reconstruction tolerance. This does not replace the 42-case physics domain: all 42 reconstructed states must still be evaluated, and the maximum of only the basis states is not the original envelope maximum. Original full F is stored unchanged; coefficients are an optional numerical reuse interface.

Rank is obtained directly from the actual active nodal force rows with tolerance s1×1e−12; it is not inferred from a squared-condition normal-equation Gram matrix. QR pivots choose actual case columns. The independent verifier imports no assembly, LP or mapping function: it reconstructs the material-frame literal targets directly from the original handoff, extracts actual physicalgroup node sets, checks every port/global force and moment, exact zero outside each case's active groups, literal metadata, actual-case basis identity and the reconstructed full force array. It also independently evaluates the direct SVD and six rigid-gauge rank.

Six integral gauge rows remove only rigid-body translations/rotations. They do not clamp a journal or establish actual bearing support. No FE solution is run in this branch. Bare journal area is not a certified bearing loading width; pressure centre, clearance, preload and load sharing remain unknown. Static-stop engagement remains restricted to the original neutral cases and requires unload/release. Real mounting bolts, pin contact, friction, material certification, fatigue and whole-robot loads remain separate.

Reproduction uses the repository .venv/bin/python. For each dataset, invoke assemble_load_envelope.py with its report mesh_path, CAD_source_path and physicalgroup_report_path as --mesh, --cad-source and --port-report, plus --prefix. Then run verify_load_envelope.py with all dataset prefixes, followed by freeze_load_envelope.py. manifest.json records exact executable commands and all input/output SHA256 values. Upstream frozen data is read only.
'''
    (OUT/'load_envelope_review.md').write_text(text)
    commands=[]
    for prefix,r,v in datasets:
        commands.append(f".venv/bin/python .scratch/gorilla_internal_g8_load_envelope/assemble_load_envelope.py --mesh {Path(r['mesh_path']).relative_to(ROOT)} --cad-source {Path(r['CAD_source_path']).relative_to(ROOT)} --port-report {Path(r['physicalgroup_report_path']).relative_to(ROOT)} --prefix {prefix}")
    commands.append('.venv/bin/python .scratch/gorilla_internal_g8_load_envelope/verify_load_envelope.py '+' '.join(p for p,_,_ in datasets))
    commands.append('.venv/bin/python .scratch/gorilla_internal_g8_load_envelope/freeze_load_envelope.py')
    outputs=[p for p in sorted(OUT.iterdir()) if p.is_file() and p.name!='manifest.json']
    manifest={'schema':'G8_full_literal_load_envelope_snapshot','frozen':True,'dataset_prefixes':[p for p,_,_ in datasets],
              'inputs':[{'path':str(p.resolve()),'sha256':sha(p)} for p in sorted(inputs)],
              'outputs':[{'path':str(p.resolve()),'sha256':sha(p),'bytes':p.stat().st_size} for p in outputs],
              'reproduction_commands':commands,'same_original42_no_domain_shrink':True,'solver_executed':False,'physical_release':False,
              'scope':'Actual tagged CAD full-wrench conditional nodal-load envelope and exact actual-case linear basis; no contact or FE qualification.'}
    save(OUT/'manifest.json',manifest)
    print(json.dumps({'datasets':manifest['dataset_prefixes'],'outputs':len(outputs),'manifest_sha256':sha(OUT/'manifest.json')},indent=2))
if __name__=='__main__':main()

"""Native triangle orthographic z-buffer; all new G14 material retained, no shape editing."""
from pathlib import Path
import hashlib,json,math,gzip,argparse
import numpy as np
from PIL import Image,ImageDraw,ImageFont

O=Path(__file__).resolve().parent
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
ap=argparse.ArgumentParser();ap.add_argument('--variant',choices=['a','b'],required=True);args=ap.parse_args();O=O/args.variant
ROOT=next(p for p in O.parents if (p/'sai_robots_engineering_rules.md').is_file())
S=O/'scene.json.gz';s=json.loads(gzip.decompress(S.read_bytes()));C=np.array([0.,0.,2.25])
context=ROOT/'.scratch/gorilla_internal_g9_whole_source/scene.json.gz';whole=json.loads(gzip.decompress(context.read_bytes()))
keep=set();parts=s['parts']+[p for p in whole['parts'] if p.get('role') in ('armor_surface','armor_cover')]; C=np.array([0.,0.,2.23])
for p in parts:
    p['faces']=[[f[0],f[i],f[i+1]] for f in p['faces'] for i in range(1,len(f)-1)]
W,H=1000,950;PPM=560.
font=ImageFont.truetype('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf',19)
small=ImageFont.truetype('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf',14)
views={'front':([1.,0.,0.],[0.,0.,1.],[0.,-1.,0.]),
       'left':([0.,1.,0.],[0.,0.,1.],[-1.,0.,0.]),
       'rear':([-1.,0.,0.],[0.,0.,1.],[0.,1.,0.]),
       'top':([0.,0.,1.],[1.,0.,0.],[0.,-1.,0.]),
       'threequarter':([.85,1.2,.50],[0.,0.,1.],None)}

def draw_native(po,key):
    cam,up,right=views[key];cam=np.asarray(cam);cam/=np.linalg.norm(cam);up=np.asarray(up)
    if right is None:
        right=np.cross(up,cam);right/=np.linalg.norm(right);up=np.cross(cam,right)
    else:right=np.asarray(right)
    target=C
    depth=np.full((H,W),-np.inf);rgb=np.full((H,W,3),255,np.uint8)
    light=np.array([1.,-.5,1.4]);light/=np.linalg.norm(light)
    for p in po['parts']:
        v=np.asarray(p['vertices_world_m']);vv=v-target
        q=np.column_stack([W/2+vv@right*PPM,H/2-vv@up*PPM,vv@cam])
        triangles=np.asarray(p['faces']);coords=q[triangles]
        normal=np.cross(v[triangles[:,1]]-v[triangles[:,0]],v[triangles[:,2]]-v[triangles[:,0]])
        norm=np.linalg.norm(normal,axis=1);valid=norm>1e-15;normal[valid]/=norm[valid,None]
        if p.get('density_kg_m3') is not None:color=np.array([114,142,149])
        else:color=np.array(p.get('rgba',[.60,.56,.42,1.])[:3])*255
        for i,a in enumerate(coords):
            if not valid[i] or normal[i]@cam<=0:continue
            xa,ya,za=a[0];xb,yb,zb=a[1];xc,yc,zc=a[2]
            den=(yb-yc)*(xa-xc)+(xc-xb)*(ya-yc)
            if abs(den)<1e-9:continue
            x0=max(0,int(math.floor(a[:,0].min())));x1=min(W-1,int(math.ceil(a[:,0].max())))
            y0=max(80,int(math.floor(a[:,1].min())));y1=min(H-88,int(math.ceil(a[:,1].max())))
            if x1<x0 or y1<y0:continue
            xx,yy=np.meshgrid(np.arange(x0,x1+1)+.5,np.arange(y0,y1+1)+.5)
            aa=((yb-yc)*(xx-xc)+(xc-xb)*(yy-yc))/den
            bb=((yc-ya)*(xx-xc)+(xa-xc)*(yy-yc))/den;cc=1-aa-bb
            zz=aa*za+bb*zb+cc*zc;z=depth[y0:y1+1,x0:x1+1]
            mask=(aa>=-1e-9)&(bb>=-1e-9)&(cc>=-1e-9)&(zz>z)
            shade=.67+.33*max(0,float(normal[i]@light));col=np.asarray(color*shade,np.uint8)
            z[mask]=zz[mask];rgb[y0:y1+1,x0:x1+1][mask]=col
    im=Image.fromarray(rgb);d=ImageDraw.Draw(im)
    d.text((24,17),'GORILLA G14 SHOULDER '+args.variant.upper()+' / '+key.upper()+' / '+po['id'],font=font,fill=(30,39,46))
    d.text((24,46),'Finite CSG/OCC assembly + complete OEM references; internal candidate only.',font=small,fill=(65,73,81))
    d.text((24,H-77),'Fixed reaction / fast input / moving output have separate owners.',font=small,fill=(40,45,50))
    d.text((24,H-53),'Candidate diagnostic; original armor and actual connection remain unaccepted.',font=small,fill=(40,45,50))
    d.line((W-161,H-26,W-49,H-26),fill=(20,25,30),width=3)
    d.text((W-150,H-49),'200 mm',font=small,fill=(20,25,30))
    d.text((24,H-28),'Source '+sha(S)[:16]+' / same 560 px/m',font=small,fill=(40,45,50))
    path=O/(po['id']+'_'+key+'.png');im.save(path)
    return {'path':str(path),'sha256':sha(path),'source_sha256':sha(S),'pose':po['id'],'camera':key,
        'camera_direction':cam.tolist(),'up':up.tolist(),'right':np.asarray(right).tolist(),'target_world_m':np.asarray(target).tolist(),
        'pixels_per_m':PPM,'dimensions_px':[W,H],'hidden_parts':[],'projection':'native triangle z-buffer orthographic','physical_acceptance':False}

outputs=[]
neutral={'id':'original_armor_context','parts':parts}
for key in ['front','left','rear','top']:outputs.append(draw_native(neutral,key))

im=Image.new('RGB',(W*2,H*2),'white')
for i,p in enumerate(outputs[:4]):im.paste(Image.open(p['path']),(i%2*W,i//2*H))
contact=O/'native_armor_context_four_view.png';im.save(contact)
outputs.append({'path':str(contact),'sha256':sha(contact),'source_sha256':sha(S),'operation':'unaltered panel assembly; no shape/color edits'})
(O/'native_armor_context_render_manifest.json').write_text(json.dumps({'robot_id':'gorilla_v0_1','source_sha256':sha(S),'renderer_sha256':sha(Path(__file__)),
    'outputs':outputs,'original_armor_context_sha256':sha(context),'source_part_count':len(parts),'new_part_count':len(s['parts']),'original_armor_part_count':len(parts)-len(s['parts']),'hidden_parts':[p['name'] for p in whole['parts'] if p.get('role') not in ('armor_surface','armor_cover')], 'hidden_reason':'All 128 new parts and all original armor role parts are displayed at exactly the bare-view scale. Original armor source is independently bound. Cropping by camera is explicit; other whole modules omitted from context image, nine groups are queried separately by check_report.json.','scope':'Internal shoulder/torso candidate only; 128 new parts + all original armor-role parts without selective geometry hiding; no exterior or physical acceptance' },indent=2)+'\n')
print('RENDERED',len(outputs),flush=True)

from pathlib import Path
import urllib.request,hashlib,json,subprocess,re,time
O=Path(__file__).resolve().parent
urls={'hydraforce_hsp10_47c.pdf':'https://www.hydraforce.com/globalassets/2.-products/product-pdf-files/hsp10-47c_25sep2024_16-40.pdf','hydraforce_hsp12_product.html':'https://www.hydraforce.com/products/cartridge-valves/electro-proportional-valves/hsp12-47c/','hydraforce_coil10_e_product.html':'https://www.hydraforce.com/products/cartridge-valves/cartridge-valve-accessories/coil_10-xxexx/'}
r=[]
for name,url in urls.items():
 try:
  data=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=15).read();(O/name).write_bytes(data)
  r.append({'path':name,'url':url,'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest(),'pdf_magic':data.startswith(b'%PDF'),'downloaded_utc':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime())})
  if data.startswith(b'%PDF'):subprocess.run(['pdftotext','-layout',str(O/name),str(O/(name[:-4]+'.txt'))],check=True)
  if name.endswith('.html'):
   for x in sorted(set(re.findall(r'[^\s\"<>]{0,200}\.pdf',data.decode(errors='replace')))):print(name,x)
 except Exception as e:r.append({'path':name,'url':url,'error':str(e)})
(O/'primary_download_receipt_v3.json').write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r,indent=2))

"""Gorilla F2 closed boundary containment audit. Analysis copies only, original A/B untouched.

Manifold.decompose is a boundary partition, not sufficient as a material-connectivity
count when nested closed cavities have been flipped by fix_normals(). Normalization
below applies nesting-parity boundary orientation, with exact containment volumes.
No repaired copy is an accepted candidate or third layout.
"""
from pathlib import Path
import json,hashlib,copy
import numpy as np,trimesh as tr,manifold3d as mf
O=Path(__file__).resolve().parent;TH=1e-10
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
def M(p):return mf.Manifold(mf.Mesh64(np.asarray(p['vertices_world_m'],np.float64),np.array([[f[0],f[i],f[i+1]] for f in p['faces'] for i in range(1,len(f)-1)],np.uint64)))
def positive(c):
 q=c.to_mesh64();v=np.asarray(q.vert_properties[:,:3],np.float64);f=np.asarray(q.tri_verts,np.uint64)
 if c.volume()<0:f=f[:,::-1].copy()
 return mf.Manifold(mf.Mesh64(np.array(v,dtype=np.float64,order="C",copy=True),np.array(f,dtype=np.uint64,order="C",copy=True)))
def normalize(p):
 raw=M(p);cs=[c for c in raw.decompose() if abs(c.volume())>TH];ss=[positive(c) for c in cs];vs=[c.volume() for c in ss];parents=[];tests=[]
 for i,c in enumerate(ss):
  boxes=[x.bounding_box() for x in ss];a=np.array(boxes[i]);choices=[]
  for j,d in enumerate(ss):
   if i==j or vs[j]<=vs[i]+TH:continue
   b=np.array(boxes[j])
   if not(np.all(a[:3]>=b[:3]-1e-8) and np.all(a[3:]<=b[3:]+1e-8)):continue
   iv=(c^d).volume();contains=abs(iv-vs[i])<=max(TH,vs[i]*1e-6);tests.append({'inner':i,'outer':j,'inner_volume_m3':vs[i],'intersection_volume_m3':iv,'fully_contained':contains})
   if contains:choices.append(j)
  parents.append(min(choices,key=lambda j:vs[j]) if choices else None)
 depths=[]
 for i in range(len(ss)):
  dep=0;j=parents[i]
  while j is not None:dep+=1;j=parents[j]
  depths.append(dep)
 result=mf.Manifold()
 for i in sorted(range(len(ss)),key=lambda k:depths[k]):result=(result+ss[i]) if depths[i]%2==0 else (result-ss[i])
 q=result.to_mesh64();cp=copy.deepcopy(p);cp['vertices_world_m']=np.asarray(q.vert_properties[:,:3]).tolist();cp['faces']=np.asarray(q.tri_verts).tolist();cp['analysis_only_boundary_orientation_copy']=True
 row={'id':p['name'],'original_net_volume_m3':raw.volume(),'normalized_net_volume_m3':result.volume(),'volume_delta_m3':result.volume()-raw.volume(),'meaningful_boundary_components':len(cs),'containment_parents':parents,'depths':depths,'original_boundary_signed_volumes_m3':[c.volume() for c in cs],'orientation_conflict_boundary_indices':[i for i,c in enumerate(cs) if (c.volume()>0)!=(depths[i]%2==0)],'containment_tests':tests,'normalized_material_root_components':sum(d==0 for d in depths),'nested_even_material_islands':sum(d>0 and d%2==0 for d in depths),'not_material_connectivity_qualification':True}
 return cp,result,row
def main():

  print(which,'affected',len(out['layouts'][which]['affected_parts']),'mass',oldmass,newmass,'pairs',[(p['original_intersection_volume_cm3'],p['normalized_intersection_volume_cm3']) for p in pairs],flush=True)
 (O/'closed_cavity_receipt.json').write_text(json.dumps(out,indent=2)+'\n')

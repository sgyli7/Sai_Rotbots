def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def read(p):
    p = Path(p)
    return json.loads(gzip.decompress(p.read_bytes()) if p.suffix == '.gz' else p.read_bytes())
def write(name, x):
    raw = (json.dumps(x, indent=2, allow_nan=False) + '\n').encode()
    p = D / name
    p.write_bytes(gzip.compress(raw, mtime=0) if p.suffix == '.gz' else raw)
def geometry(p): return (p['vertices_world_m'], p['faces'], p.get('body'))
def mesh(p):
    v = np.asarray(p['vertices_world_m'], float)
    f = np.asarray([[a[0], a[k], a[k+1]] for a in p['faces'] for k in range(1, len(a)-1)], np.uint64)
    return trimesh.Trimesh(v, f, process=False)
def transform(old, new, ids):
    if not ids or any(old[n]['faces'] != new[n]['faces'] or
            len(old[n]['vertices_world_m']) != len(new[n]['vertices_world_m']) for n in ids):
        return {'pass': False, 'reason': 'indexed geometry changed or absent'}
    a = np.concatenate([np.asarray(old[n]['vertices_world_m'], float) for n in ids])
    b = np.concatenate([np.asarray(new[n]['vertices_world_m'], float) for n in ids])
    aa, bb = a.mean(0), b.mean(0)
    u, _, vh = np.linalg.svd((a-aa).T @ (b-bb))
    rot = vh.T @ u.T
    if np.linalg.det(rot) < 0:
        vh[-1] *= -1
        rot = vh.T @ u.T
    off = bb - rot @ aa
    err = float(np.max(np.linalg.norm(a @ rot.T + off - b, axis=1)))
    T = np.eye(4); T[:3,:3] = rot; T[:3,3] = off
    return {'pass': err < 1e-7, 'maximum_indexed_vertex_error_m': err,
            'detR': float(np.linalg.det(rot)), 'T_world_new_from_old': T.tolist()}
